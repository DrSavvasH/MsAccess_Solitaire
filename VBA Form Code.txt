Private Sub c1_DblClick(Cancel As Integer)
DoCmd.GoToRecord acDataForm, "Deck", acGoTo, 1

c1.FontName = "Playing Cards"
c1.Caption = cardFont
c1.ForeColor = RGB(0, 0, 0)
    If Mid([Card], 2, 1) = "H" Then c1.ForeColor = RGB(255, 0, 0)
    If Mid([Card], 2, 1) = "D" Then c1.ForeColor = RGB(255, 0, 0)
    If Mid([Card], 3, 1) = "H" Then c1.ForeColor = RGB(255, 0, 0)
    If Mid([Card], 3, 1) = "D" Then c1.ForeColor = RGB(255, 0, 0)
    If Mid([Card], 1, 2) = "J2" Then c1.ForeColor = RGB(255, 0, 0)
    Hidden = 0
    Exit Sub
    End If
If Hidden = 0 Then
    c1.FontName = "Greek Column"
    c1.Caption = Chr(87)
    c1.ForeColor = RGB(0, 0, 0)
    Hidden = -1
    End If
End Sub

Private Sub c1_MouseDown(Button As Integer, Shift As Integer, X As Single, Y As Single)
    XPos = X
    Ypos = Y
    mseDown = True
End Sub

Private Sub c1_MouseMove(Button As Integer, Shift As Integer, X As Single, Y As Single)
Dim newLeft As Integer
Dim newTop As Integer

    'move horizontally
    If mseDown Then
        With c1
            newLeft = .Left
            newTop = .Top
            'check within sizing parameters
            If .Left + X - XPos > 0 And .Left + .Width + X - XPos < InsideWidth Then newLeft = .Left + X - XPos
            If .Top + Y - Ypos > 0 And .Top + .Height + Y - Ypos < InsideHeight Then newTop = .Top + Y - Ypos
            .Move newLeft, newTop, .Width, .Height
        End With
    End If

End Sub

            .Move newLeft, newTop, .Width, .Height
        End With
    End If
